cmake_minimum_required(VERSION 3.16)
project(software3d)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(glfw3 3.3 REQUIRED)
find_package(OpenGL REQUIRED)

set(SOURCES
    src/main.cpp
    src/glad.c
    src/framebuffer.cpp
    src/rasterizer.cpp
    src/math/vec2.cpp   
    src/math/vec3.cpp
    src/math/vec4.cpp
    src/math/mat4.cpp
    src/mesh/obj_loader.cpp
)

add_executable(software3d ${SOURCES})

target_include_directories(software3d PRIVATE include)

target_link_libraries(software3d PRIVATE glfw OpenGL::GL)

add_custom_command(TARGET software3d POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
    ${CMAKE_SOURCE_DIR}/assets
    ${CMAKE_CURRENT_BINARY_DIR}/assets
)
